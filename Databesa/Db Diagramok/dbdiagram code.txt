//Emberek


Table user {
  id integer(11) [primary key]
  image_id integer(11) [null] // Ha NULL, akkor a frontend a monogramot (pl. "PP") mutatja!
  username varchar(100)
  email varchar(150) [unique]
  password_hash varchar(255)
  created_at timestamp
  deleted_at timestamp
  update_at timestamp
  is_active tinyint
  is_deleted tinyint
}

Table trainer {
  id integer(11) [primary key]
  firstname varchar(100) 
  lastname varchar(100)
  phone_number varchar(50)
  email varchar(150)
  title text
  created_at timestamp
  deleted_at timestamp
  update_at timestamp
  is_active tinyint
  is_deleted tinyint

}

Table admin {
  id integer(11) [primary key]
  image_id integer(11) [null] // Ugyanúgy rámutathat az images táblára
  email varchar(150)
  password_hash varchar(255)
  username varchar(100)
  created_at timestamp
  deleted_at timestamp
  update_at timestamp
  is_active tinyint
  is_deleted tinyint
}

//--------------------------------------------------------------------//
//Média / Képek

Table images {
  id integer(11) [primary key]
  image_url varchar(255) [not null]
  created_at timestamp
  updated_at timestamp
}

//--------------------------------------------------------------------//
//Speciálisabb dolgok

Table user_tokens {
  id integer(11) [primary key]
  user_id integer(11) [null]   // Lehet NULL, ha adminról van szó
  admin_id integer(11) [null]  // ÚJ MEZŐ: Lehet NULL, ha userről van szó

  remember_token varchar(100)
  reset_password_token varchar(255) // Jelszóvisszaállító linkhez
  reset_password_expires datetime // Meddig érvényes a visszaállító link
  email_verified_at datetime  // Opcionális: Mikor erősítette meg az e-mail címét

  created_at timestamp
  updated_at timestamp
}

Table log {
  id integer(11) [primary key]  
  user_id integer(11)
  admin_id integer(11)
  action varchar(100)
  entity_table varchar(100)
  entity_id integer(11)
  created_at timestamp
  
}

Table contact_messages { 
  id integer(11) [primary key]
  name varchar(150)
  email varchar(150)
  message text
  
  status enum('new', 'in_progress', 'resolved') [default: 'new']
  created_at timestamp
  Resolved_at timestamp // Mikor válaszolt rá az admin
  admin_id integer(11) // Melyik admin kezelte (opcionális)
}

//--------------------------------------------------------------------//
//Vásárlás


Table memberships {
  id integer(11) [primary key]
  duration_days integer // Example. 1 (day), 30 (days), 365 (days)
  title text [not null]
  price decimal(10,2) [not null]

}

Table user_Memberships {
  id integer(11) [primary key]
  user_id integer(11)
  membership_id integer(11)
  pays_id integer(11)
  start_date datetime
  end_date datetime
  status enum('pending_payment', 'active', 'expired', 'cancelled') [default: 'pending_payment']
  created_at timestamp
}

Table pays {
  id integer(11) [primary key]
  user_id integer(11)
  membership_id integer(11)
  price decimal(10,2)
  status enum('unpaid', 'paid', 'cancelled') [default: 'unpaid']
  email_sent boolean [default: false] // Kiküldte-e a rendszer a visszaigazoló emailt
  created_at timestamp
  paid_at datetime // Akkor töltődik ki, amikor a recepciós átveszi a pénzt
}

//--------------------------------------------------------------------//
//Gym info

Table gym_info { 
  id integer(11) [primary key]
  gym_name varchar(100)
  address varchar(255)
  phone_number varchar(50)
  premium_machine_number integer
  email varchar(150)
  opening_hours text
  updated_at timestamp
  users_number integer 
  trainers_number integer

}

//Egyéni------

Table trainer_schedules {
  id integer(11) [primary key]
  trainer_id integer(11) [not null]

  //Számok - day_of_week
  day_of_week enum('monday', 'tuesday', 'wednesday', 'thursday', 'friday', 'saturday', 'sunday') [not null]
  start_time time [not null]
  end_time time [not null]
  created_at timestamp
  updated_at timestamp
}


//Csoportos------

Table classes {
  id integer(11) [primary key]
  name enum('gerinctorna', 'joga', 'noi_kondicionalo_torna', 'cross_training', 'latin_fitness', 'alakformalo') [not null]
  trainer_id integer
  description text
  duration_minutes varchar(50) [default: '45-60 perc']
  difficulty_level varchar(50) [default: 'vegyes']
  room_type enum('A_terem', 'B_terem', 'kuzdoter')
  created_at timestamp
}

Table class_bookings {
  id integer(11) [primary key]
  user_id integer(11) [not null]
  class_schedule_id integer(11) [not null] // Cserélve class_id-ról erre!
  booking_date datetime [not null]        // Melyik napra szól pontosan a foglalás
  created_at timestamp
}

Table class_schedules {
  id integer(11) [primary key]
  class_id integer(11) [not null] 
  day_of_week integer [not null] 
  start_time time [not null]     
  end_time time [not null]        
  created_at timestamp
}
